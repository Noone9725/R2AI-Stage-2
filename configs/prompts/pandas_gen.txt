Ví dụ 3: So sánh nhóm nhiều công ty (df1, df2, ..., dfN): tính trung vị và tỷ trọng
```python
import numpy as np
def get_val(df, kw):
    s = df[df['item'].str.contains(kw, case=False, na=False)]
    return float(s['value'].iloc[0]) if not s.empty else 0.0

# Lặp qua các bảng của từng công ty
invs = [get_val(df, 'Hàng tồn kho') for df in [df1, df2, df3, df4, df5, df6, df7]]
liabs = [get_val(df, 'Nợ ngắn hạn') for df in [df1, df2, df3, df4, df5, df6, df7]]
ratios = [inv / liab if liab != 0 else 0.0 for inv, liab in zip(invs, liabs)]

med = float(np.median(ratios))
top_liab = sum(l for r, l in zip(ratios, liabs) if r > med)
total_liab = sum(liabs)
result = float((top_liab / total_liab) * 100.0) if total_liab != 0 else 0.0
```
